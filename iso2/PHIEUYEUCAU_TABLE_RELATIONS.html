Bảng nguồn Bảng liên kết Điều kiện JOIN (ON Clause) Mục đích & Dữ liệu hiển thị
hososcbd_iso (h) donvi_iso (d) h.madv = d.madv Lấy tên đơn vị yêu cầu bảo dưỡng (tendv).
hososcbd_iso (h) thietbi_iso (t) h.mavt = t.mavt AND h.somay = t.somay Lấy tên thiết bị (tenvt), model máy và ID gốc (t.stt).
thietbi_iso (t) ke_hoach_bao_duong_dinh_ky_iso (k) t.stt = k.thietbi_id AND k.nam = YEAR(h.ngayyc) Lấy kế hoạch và trạng thái hoàn thành bảo dưỡng định kỳ các quý (Q1, Q2, Q3, Q4) trong năm yêu cầu.
thietbi_iso (t) thietbihckd_iso (thckd) (t.mavt = thckd.mavattu AND t.somay = thckd.somay)
OR (CONCAT(t.mavt,'-',t.somay) = thckd.mavattu AND t.somay = thckd.somay) 		Ánh xạ thiết bị bảo dưỡng sang bảng quản lý hiệu chuẩn / kiểm định.
thietbihckd_iso (thckd) kehoach_kiemdinh_2026_iso (kd) thckd.stt = kd.stt AND kd.nam_kehoach = 2026 Lấy các tháng dự kiến kiểm định trong kế hoạch năm 2026 (thang_thuchien, thang_dot2).
thietbihckd_iso (thckd) hosohckd_iso (hc) (thckd.mavattu = hc.tenmay OR thckd.somay = hc.tenmay)
AND YEAR(hc.ngayhc) = 2026 		Lấy các tháng thực tế đã thực hiện hiệu chuẩn kiểm định trong năm 2026 (MONTH(hc.ngayhc)).
Ghi chú quan trọng: Trong hệ thống, một phiếu yêu cầu (ví dụ: phieu = '2122') có thể chứa nhiều thiết bị, mỗi thiết bị tương ứng với 1 dòng trong bảng hososcbd_iso có chung số phiếu.
4. Các câu lệnh SQL thực thi trong PhieuYeuCau::getPhieuDetail()

Query 1: Thông tin tổng hợp phiếu

SELECT 
    h.phieu,
    MIN(h.ngayyc) as ngayyc,
    h.madv,
    d.tendv,
    h.ngyeucau,
    h.ngnhyeucau,
    h.dienthoai,
    h.nhomsc,
    h.cv,
    h.ycthemkh,
    COUNT(*) as so_thietbi,
    SUM(CASE WHEN h.ngayth IS NULL OR h.ngayth = '0000-00-00' THEN 1 ELSE 0 END) as tb_chuath,
    SUM(CASE WHEN h.ngayth IS NOT NULL AND h.ngayth != '0000-00-00' AND (h.ngaykt IS NULL OR h.ngaykt = '0000-00-00') THEN 1 ELSE 0 END) as tb_danglam,
    SUM(CASE WHEN h.ngaykt IS NOT NULL AND h.ngaykt != '0000-00-00' AND h.bg = 0 THEN 1 ELSE 0 END) as tb_hoanthanh,
    SUM(CASE WHEN h.bg = 1 THEN 1 ELSE 0 END) as tb_dabg
FROM hososcbd_iso h
LEFT JOIN donvi_iso d ON h.madv = d.madv
WHERE h.phieu = '2122'
GROUP BY h.phieu, h.madv, d.tendv, h.ngyeucau, 
         h.ngnhyeucau, h.dienthoai, h.nhomsc, h.cv, h.ycthemkh;

Query 2: Danh sách thiết bị và liên kết BDDK, HC/KĐ

SELECT h.*, t.tenvt, t.stt as thietbi_stt,
       thckd.stt as thckd_stt,
       GROUP_CONCAT(DISTINCT kd.thang_thuchien ORDER BY kd.thang_thuchien) as planned_months,
       GROUP_CONCAT(DISTINCT kd.thang_dot2 ORDER BY kd.thang_dot2) as planned_months_dot2,
       GROUP_CONCAT(DISTINCT MONTH(hc.ngayhc) ORDER BY hc.ngayhc) as inspected_months,
       GROUP_CONCAT(DISTINCT 
           CONCAT(
               IF((k.qui_1 IS NOT NULL AND k.qui_1 != '') OR k.qui_1_hoantat = 1, CONCAT('Q1:', COALESCE(k.qui_1_hoantat, 0), ','), ''),
               IF((k.qui_2 IS NOT NULL AND k.qui_2 != '') OR k.qui_2_hoantat = 1, CONCAT('Q2:', COALESCE(k.qui_2_hoantat, 0), ','), ''),
               IF((k.qui_3 IS NOT NULL AND k.qui_3 != '') OR k.qui_3_hoantat = 1, CONCAT('Q3:', COALESCE(k.qui_3_hoantat, 0), ','), ''),
               IF((k.qui_4 IS NOT NULL AND k.qui_4 != '') OR k.qui_4_hoantat = 1, CONCAT('Q4:', COALESCE(k.qui_4_hoantat, 0), ','), '')
           )
       ) as bddk_quarters_raw
FROM hososcbd_iso h
LEFT JOIN thietbi_iso t ON h.mavt = t.mavt AND h.somay = t.somay
LEFT JOIN thietbihckd_iso thckd ON (
    (t.mavt = thckd.mavattu AND t.somay = thckd.somay)
    OR (CONCAT(t.mavt, '-', t.somay) = thckd.mavattu AND t.somay = thckd.somay)
)
LEFT JOIN ke_hoach_bao_duong_dinh_ky_iso k ON t.stt = k.thietbi_id AND k.nam = YEAR(h.ngayyc)
LEFT JOIN kehoach_kiemdinh_2026_iso kd ON thckd.stt = kd.stt AND kd.nam_kehoach = 2026
LEFT JOIN hosohckd_iso hc ON (thckd.mavattu = hc.tenmay OR thckd.somay = hc.tenmay) AND YEAR(hc.ngayhc) = 2026
WHERE h.phieu = '2122'
GROUP BY h.stt
ORDER BY h.stt ASC;
