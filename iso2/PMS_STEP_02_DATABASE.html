1. Hybrid Database Strategy

🤔 Tại Sao Cần Cả PostgreSQL VÀ MongoDB?

🐘 PostgreSQL (Structured)

Dùng cho:

  • ✅ Core business data (users, plans, tasks)
  • ✅ Dữ liệu có relationships chặt chẽ
  • ✅ Cần ACID transactions
  • ✅ Queries phức tạp (JOINs, aggregations)
  • ✅ Reporting & analytics

Ví dụ: Thông tin người dùng, kế hoạch sản xuất, nhiệm vụ

🍃 MongoDB (Dynamic)

Dùng cho:

  • ✅ Cấu hình động theo từng Enterprise
  • ✅ Custom fields & workflows
  • ✅ Logs & audit trails
  • ✅ Temporary data & caching
  • ✅ Dữ liệu không cố định schema

Ví dụ: Enterprise configs, custom form fields, activity logs

4. Laravel Models với Dual Connections

🐘 PostgreSQL Models

// app/Models/Enterprise.php namespace App\Models; use Illuminate\Database\Eloquent\Model; use Illuminate\Database\Eloquent\SoftDeletes; class Enterprise extends Model { use SoftDeletes; protected $connection = 'pgsql'; protected $fillable = [ 'name', 'code', 'status', 'description' ]; public function users() { return $this->hasMany(User::class); } public function plans() { return $this->hasMany(ProductionPlan::class); } // Get config from MongoDB public function config() { return EnterpriseConfig::where('enterprise_id', $this->id)->first(); } }
// app/Models/Task.php namespace App\Models; use Illuminate\Database\Eloquent\Model; use Illuminate\Database\Eloquent\SoftDeletes; class Task extends Model { use SoftDeletes; protected $connection = 'pgsql'; protected $fillable = [ 'plan_id', 'name', 'description', 'assigned_to', 'status', 'priority', 'start_date', 'end_date', 'estimated_hours', 'actual_hours', 'progress', 'order' ]; protected $casts = [ 'start_date' => 'date', 'end_date' => 'date', ]; public function plan() { return $this->belongsTo(ProductionPlan::class, 'plan_id'); } public function assignedUser() { return $this->belongsTo(User::class, 'assigned_to'); } // Get custom data from MongoDB public function details() { return TaskDetail::where('task_id', $this->id)->first(); } }
