1. Laravel Sanctum - Authentication

📦 Setup Sanctum

# 1. Install Sanctum (đã cài ở Bước 1) composer require laravel/sanctum # 2. Publish config php artisan vendor:publish --provider="Laravel\Sanctum\SanctumServiceProvider" # 3. Run migration php artisan migrate

⚙️ Cấu Hình User Model

// app/Models/User.php namespace App\Models; use Illuminate\Foundation\Auth\User as Authenticatable; use Laravel\Sanctum\HasApiTokens; class User extends Authenticatable { use HasApiTokens; // ← Thêm trait này protected $fillable = [ 'name', 'email', 'password', 'role_id', 'enterprise_id', 'avatar_url', 'status' ]; protected $hidden = [ 'password', 'remember_token', ]; protected $casts = [ 'password' => 'hashed', 'last_login_at' => 'datetime', ]; public function role() { return $this->belongsTo(Role::class); } public function enterprise() { return $this->belongsTo(Enterprise::class); } }

🔐 Auth Controller

// app/Http/Controllers/AuthController.php namespace App\Http\Controllers; use App\Models\User; use Illuminate\Http\Request; use Illuminate\Support\Facades\Hash; use Illuminate\Validation\ValidationException; class AuthController extends Controller { public function login(Request $request) { $request->validate([ 'email' => 'required|email', 'password' => 'required', ]); $user = User::where('email', $request->email)->first(); if (! $user || ! Hash::check($request->password, $user->password)) { throw ValidationException::withMessages([ 'email' => ['The provided credentials are incorrect.'], ]); } // Update last login $user->update(['last_login_at' => now()]); // Create token $token = $user->createToken('auth-token')->plainTextToken; return response()->json([ 'user' => $user->load(['role', 'enterprise']), 'token' => $token, ]); } public function logout(Request $request) { $request->user()->currentAccessToken()->delete(); return response()->json(['message' => 'Logged out successfully']); } public function me(Request $request) { return response()->json([ 'user' => $request->user()->load(['role', 'enterprise']) ]); } }

2. Production Plans API

📋 Request Validation

# Tạo Form Request php artisan make:request StoreProductionPlanRequest
// app/Http/Requests/StoreProductionPlanRequest.php namespace App\Http\Requests; use Illuminate\Foundation\Http\FormRequest; class StoreProductionPlanRequest extends FormRequest { public function authorize() { return true; // Or check permissions } public function rules() { return [ 'name' => 'required|string|max:255', 'code' => 'required|string|unique:production_plans', 'description' => 'nullable|string', 'start_date' => 'required|date', 'end_date' => 'required|date|after:start_date', 'status' => 'in:draft,planning,in_progress,completed,cancelled', ]; } }

🎯 Controller

// app/Http/Controllers/ProductionPlanController.php namespace App\Http\Controllers; use App\Models\ProductionPlan; use App\Http\Requests\StoreProductionPlanRequest; use Illuminate\Http\Request; class ProductionPlanController extends Controller { public function index(Request $request) { $query = ProductionPlan::with(['enterprise', 'createdBy']); // Filter by enterprise (if not admin) if ($request->user()->role->slug !== 'admin') { $query->where('enterprise_id', $request->user()->enterprise_id); } // Filters if ($request->has('status')) { $query->where('status', $request->status); } // Pagination $plans = $query->paginate(20); return response()->json($plans); } public function store(StoreProductionPlanRequest $request) { $plan = ProductionPlan::create([ ...$request->validated(), 'enterprise_id' => $request->user()->enterprise_id, 'created_by' => $request->user()->id, ]); return response()->json($plan->load(['enterprise', 'createdBy']), 201); } public function show($id) { $plan = ProductionPlan::with(['enterprise', 'createdBy', 'tasks']) ->findOrFail($id); return response()->json($plan); } public function update(Request $request, $id) { $plan = ProductionPlan::findOrFail($id); $plan->update($request->validated()); return response()->json($plan->load(['enterprise', 'createdBy'])); } public function destroy($id) { $plan = ProductionPlan::findOrFail($id); $plan->delete(); return response()->json(['message' => 'Plan deleted']); } }

3. Tasks API (với MongoDB)

✅ Task Controller

// app/Http/Controllers/TaskController.php namespace App\Http\Controllers; use App\Models\Task; use App\Models\TaskDetail; use App\Events\TaskUpdated; use Illuminate\Http\Request; class TaskController extends Controller { public function index(Request $request) { $query = Task::with(['plan', 'assignedUser']); // Filter by plan if ($request->has('plan_id')) { $query->where('plan_id', $request->plan_id); } // Filter by status if ($request->has('status')) { $query->where('status', $request->status); } // Filter by assigned user if ($request->has('assigned_to')) { $query->where('assigned_to', $request->assigned_to); } $tasks = $query->orderBy('order')->get(); return response()->json($tasks); } public function store(Request $request) { $validated = $request->validate([ 'plan_id' => 'required|exists:production_plans,id', 'name' => 'required|string|max:255', 'description' => 'nullable|string', 'assigned_to' => 'nullable|exists:users,id', 'status' => 'in:todo,in_progress,review,done', 'priority' => 'in:low,medium,high,urgent', 'start_date' => 'nullable|date', 'end_date' => 'nullable|date', 'custom_data' => 'nullable|array', // For MongoDB ]); // Create in PostgreSQL $task = Task::create($validated); // Save custom data to MongoDB if ($request->has('custom_data')) { TaskDetail::create([ 'task_id' => $task->id, 'custom_data' => $request->custom_data, ]); } return response()->json($task->load(['plan', 'assignedUser']), 201); } public function show($id) { $task = Task::with(['plan', 'assignedUser'])->findOrFail($id); // Load MongoDB details $details = TaskDetail::where('task_id', $id)->first(); return response()->json([ 'task' => $task, 'details' => $details, ]); } public function update(Request $request, $id) { $task = Task::findOrFail($id); $oldStatus = $task->status; $task->update($request->only([ 'name', 'description', 'assigned_to', 'status', 'priority', 'start_date', 'end_date', 'progress' ])); // Update MongoDB details if needed if ($request->has('custom_data')) { TaskDetail::updateOrCreate( ['task_id' => $id], ['custom_data' => $request->custom_data] ); } // Broadcast event (WebSocket) event(new TaskUpdated( $task, $oldStatus !== $task->status ? 'status_changed' : 'updated' )); return response()->json($task->load(['plan', 'assignedUser'])); } public function updateOrder(Request $request) { // Bulk update task order for Kanban drag & drop $request->validate([ 'tasks' => 'required|array', 'tasks.*.id' => 'required|exists:tasks,id', 'tasks.*.order' => 'required|integer', 'tasks.*.status' => 'required|string', ]); foreach ($request->tasks as $taskData) { Task::where('id', $taskData['id'])->update([ 'order' => $taskData['order'], 'status' => $taskData['status'], ]); } return response()->json(['message' => 'Order updated']); } public function destroy($id) { $task = Task::findOrFail($id); // Delete MongoDB details TaskDetail::where('task_id', $id)->delete(); // Delete task $task->delete(); return response()->json(['message' => 'Task deleted']); } }

4. File Upload API

// app/Http/Controllers/FileUploadController.php namespace App\Http\Controllers; use Illuminate\Http\Request; use Illuminate\Support\Facades\Storage; class FileUploadController extends Controller { public function upload(Request $request) { $request->validate([ 'file' => 'required|file|max:10240', // Max 10MB ]); $file = $request->file('file'); $filename = time() . '_' . $file->getClientOriginalName(); // Store in public disk $path = $file->storeAs('uploads', $filename, 'public'); return response()->json([ 'filename' => $filename, 'url' => Storage::url($path), 'size' => $file->getSize(), ]); } }
# Tạo symbolic link cho storage/public php artisan storage:link
